# 要在棋盘上选位置的技能：守夜人（点灯）、钓叟（收灯）
#   SW_OPENING=off SW_BOOT_WAIT=2 tools/playtest/agent.sh a7 drive tools/playtest/steps_aim.txt aim
sleep 2
call pad_adapter.gd debug_unlock_all
click_btn PlayButton
sleep 3
call pad_adapter.gd debug_pick_character 0
sleep 6
call pad_adapter.gd debug_charge 9
sleep 1
# 进了第 1 关、充能充满：aim=0 left=0 cand=0
call pad_adapter.gd debug_state
# 按空格：该进选目标（aim=1，充能还是满的）
key Space
sleep 1
call pad_adapter.gd debug_state
# 方向键换一格
key Right
sleep 1
call pad_adapter.gd debug_state
# 空格确认：放一盏（left 1，充能还满）
key Space
sleep 1
call pad_adapter.gd debug_state
# 再确认一次：放完，充能清零
key Space
sleep 1
call pad_adapter.gd debug_state
# 再按空格：没充能了，什么都不该发生
key Space
sleep 1
call pad_adapter.gd debug_state
